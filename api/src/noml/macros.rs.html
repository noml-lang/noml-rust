1//! # NOML Macros
2//!
3//! Convenient macros for creating NOML values programmatically.
4
5/// Create a NOML value using a convenient macro syntax
6///
7/// This macro allows you to create complex `Value` structures using a JSON-like syntax
8/// that's easy to read and write.
9///
10/// # Example
11///
12/// ```rust
13/// use noml::{noml_value, Value};
14///
15/// let config = noml_value!({
16///     "name" => "my-app",
17///     "version" => "1.0.0",
18///     "features" => ["parsing", "validation"],
19///     "server" => {
20///         "host" => "localhost",
21///         "port" => 8080
22///     },
23///     "debug" => true,
24///     "timeout" => 30.5
25/// });
26///
27/// assert_eq!(config.get("server.port").unwrap().as_integer().unwrap(), 8080);
28/// assert_eq!(config.get("features").unwrap().as_array().unwrap().len(), 2);
29/// ```
30///
31/// # Supported Types
32///
33/// - **Strings**: `"text"` → `Value::String`
34/// - **Integers**: `42` → `Value::Integer`  
35/// - **Floats**: `3.14` → `Value::Float`
36/// - **Booleans**: `true`, `false` → `Value::Bool`
37/// - **Arrays**: `[1, 2, 3]` → `Value::Array`
38/// - **Tables**: `{ "key" => "value" }` → `Value::Table`
39/// - **Null**: `null` → `Value::Null`
40#[macro_export]
41macro_rules! noml_value {
42    // Null value
43    (null) => {
44        $crate::Value::Null
45    };
46
47    // Boolean values
48    (true) => {
49        $crate::Value::Bool(true)
50    };
51    (false) => {
52        $crate::Value::Bool(false)
53    };
54
55    // Array values
56    ([ $($item:tt),* $(,)? ]) => {
57        $crate::Value::Array(vec![
58            $(noml_value!($item)),*
59        ])
60    };
61
62    // Table values
63    ({ $($key:tt => $value:tt),* $(,)? }) => {{
64        let mut table = std::collections::BTreeMap::new();
65        $(
66            let key_value = noml_value!($key);
67            let key_str = match key_value {
68                $crate::Value::String(s) => s,
69                _ => panic!("Table keys must be strings"),
70            };
71            table.insert(key_str, noml_value!($value));
72        )*
73        $crate::Value::Table(table)
74    }};
75
76    // String literal
77    ($val:literal) => {
78        $crate::Value::from($val)
79    };
80}

82#[cfg(test)]
83mod tests {
84    use crate::Value;
85
86    #[test]
87    fn test_noml_value_primitives() {
88        // Test basic types
89        assert_eq!(noml_value!(null), Value::Null);
90        assert_eq!(noml_value!(true), Value::Bool(true));
91        assert_eq!(noml_value!(false), Value::Bool(false));
92        assert_eq!(noml_value!("hello"), Value::String("hello".to_string()));
93        assert_eq!(noml_value!(42), Value::Integer(42));
94        assert_eq!(noml_value!(3.15), Value::Float(3.15));
95    }
96
97    #[test]
98    fn test_noml_value_array() {
99        let arr = noml_value!([1, 2, 3]);
100        if let Value::Array(values) = arr {
101            assert_eq!(values.len(), 3);
102            assert_eq!(values[0], Value::Integer(1));
103            assert_eq!(values[1], Value::Integer(2));
104            assert_eq!(values[2], Value::Integer(3));
105        } else {
106            panic!("Expected array");
107        }
108    }
109
110    #[test]
111    fn test_noml_value_mixed_array() {
112        let arr = noml_value!([1, "hello", true, null]);
113        if let Value::Array(values) = arr {
114            assert_eq!(values.len(), 4);
115            assert_eq!(values[0], Value::Integer(1));
116            assert_eq!(values[1], Value::String("hello".to_string()));
117            assert_eq!(values[2], Value::Bool(true));
118            assert_eq!(values[3], Value::Null);
119        } else {
120            panic!("Expected array");
121        }
122    }
123
124    #[test]
125    fn test_noml_value_table() {
126        let table = noml_value!({
127            "name" => "test",
128            "port" => 8080,
129            "debug" => true
130        });
131
132        if let Value::Table(map) = table {
133            assert_eq!(map.len(), 3);
134            assert_eq!(map.get("name").unwrap(), &Value::String("test".to_string()));
135            assert_eq!(map.get("port").unwrap(), &Value::Integer(8080));
136            assert_eq!(map.get("debug").unwrap(), &Value::Bool(true));
137        } else {
138            panic!("Expected table");
139        }
140    }
141
142    #[test]
143    fn test_noml_value_nested() {
144        let config = noml_value!({
145            "app" => {
146                "name" => "my-app",
147                "version" => "1.0.0"
148            },
149            "features" => ["parsing", "validation"],
150            "server" => {
151                "host" => "localhost",
152                "port" => 8080
153            }
154        });
155
156        // Test accessing nested values
157        assert_eq!(
158            config.get("app.name").unwrap().as_string().unwrap(),
159            "my-app"
160        );
161        assert_eq!(
162            config.get("server.port").unwrap().as_integer().unwrap(),
163            8080
164        );
165
166        let features = config.get("features").unwrap().as_array().unwrap();
167        assert_eq!(features.len(), 2);
168        assert_eq!(features[0].as_string().unwrap(), "parsing");
169    }
170}
